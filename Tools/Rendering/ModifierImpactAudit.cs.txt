    }

guard.SetActive(true);

var motion = (UnityEngine.UI.Toggle)data.FindProperty("reducedMotionToggle").objectReferenceValue; motion.SetIsOnWithoutNotify(false);
((UnityEngine.UI.Toggle)data.FindProperty("soundToggle").objectReferenceValue).SetIsOnWithoutNotify(false);

var cards = (System.Collections.Generic.List<TackleRigCardView>)typeof(TackleLandingPresentation).GetField("cards", hidden).GetValue(view);
foreach (var pair in new[] { new[] { "phase", "SHALLOWS / LAND" }, new[] { "haul", "HAUL " + rig.HaulValue }, new[] { "creature", "SARDINE" }, new[] { "tags", "SCHOOLING" }, new[] { "resistance", "2000" }, new[] { "movement", model.Pattern.ToString().ToUpperInvariant() }, new[] { "message", "RIG ACTIVATING  /  SPACE TO FINISH" } })

var portrait = (UnityEngine.UI.Image)viewData.FindProperty("portrait").objectReferenceValue;

var track = (UnityEngine.RectTransform)viewData.FindProperty("track").objectReferenceValue;
foreach (string field in new[] { "fish", "goodZone", "perfectZone" }) {
    var rect = (UnityEngine.RectTransform)viewData.FindProperty(field).objectReferenceValue;
    rect.anchoredPosition = new UnityEngine.Vector2(((float)(field == "fish" ? model.FishPosition : model.ZonePosition) - .5f) * track.rect.width, 0);
    double width = field == "fish" ? model.FishWidth : field == "goodZone" ? model.GoodWidth : model.PerfectWidth;
    rect.sizeDelta = new UnityEngine.Vector2((float)width * track.rect.width, rect.sizeDelta.y);
}
typeof(TackleLandingPresentation).GetMethod("PrepareLamps", hidden).Invoke(view, new object[] { model.InitialAttempts });
var lamps = (System.Collections.Generic.List<UnityEngine.UI.Image>)typeof(TackleLandingPresentation).GetField("lamps", hidden).GetValue(view);
for (int i = 0; i < lamps.Count; i++) lamps[i].sprite = (UnityEngine.Sprite)viewData.FindProperty(i < model.AttemptsRemaining ? "lampOn" : "lampOff").objectReferenceValue;
((TMPro.TMP_Text)viewData.FindProperty("attempts").objectReferenceValue).text = model.AttemptsRemaining + " ATTEMPTS LEFT";
var feedback = root.GetComponent<LandingStrikeFeedback>(); feedback.Present(receipt, cards, true, true);
var number = (TMPro.TMP_Text)data.FindProperty("value").objectReferenceValue;
var operation = (TMPro.TMP_Text)data.FindProperty("operation").objectReferenceValue;
var particles = (LandingRushGraphic)data.FindProperty("particles").objectReferenceValue;
var machine = (UnityEngine.RectTransform)data.FindProperty("machine").objectReferenceValue;
var scoringPortrait = root.GetComponent<LandingScoringPortrait>();
var machineRest = machine.anchoredPosition; var portraitRest = portrait.rectTransform.localScale; var operationRest = operation.rectTransform.localScale;
var fish = (UnityEngine.RectTransform)viewData.FindProperty("fish").objectReferenceValue; var fishRest = fish.anchoredPosition;
var cardScales = System.Linq.Enumerable.ToArray(System.Linq.Enumerable.Select(cards, card => card.transform.localScale));
System.Action<float> sample = time => {
    typeof(LandingScoreRush).GetMethod("Render", hidden).Invoke(rush, new object[] { time });
    ((TMPro.TMP_Text)viewData.FindProperty("points").objectReferenceValue).text = LandingScoreRush.Format(rush.DisplayedTotal);
    ((UnityEngine.RectTransform)viewData.FindProperty("progressFill").objectReferenceValue).anchorMax = new UnityEngine.Vector2(UnityEngine.Mathf.Clamp01((float)(rush.DisplayedTotal / model.Resistance)), 1);
};

    var helper = new UnityEngine.UI.VertexHelper(); try { typeof(LandingRushGraphic).GetMethod("OnPopulateMesh", hidden | System.Reflection.BindingFlags.DeclaredOnly).Invoke(particles, new object[] { helper }); return helper.currentVertCount; }
    finally { helper.Dispose(); }
};
var cameraObject = new UnityEngine.GameObject("Private modifier capture", typeof(UnityEngine.Camera));
var camera = cameraObject.GetComponent<UnityEngine.Camera>(); camera.enabled = false; camera.orthographic = true; camera.orthographicSize = 1080; camera.transform.position = new UnityEngine.Vector3(0, 0, -1000);
var target = new UnityEngine.RenderTexture(1920, 1080, 24); var previous = UnityEngine.RenderTexture.active;

    UnityEngine.Canvas.ForceUpdateCanvases();

};

{

    check(!particles.raycastTarget, "Modifier geometry intercepts input.");

    foreach (var step in System.Linq.Enumerable.Skip(sequence.Steps, 1))
    {
        sample((step.Time + step.LaunchAt) * .5f);
        check(rush.DisplayedStrike == step.PreviousValue, "Preparation exposed future points.");

            .Append(",\"launch\":").Append(step.LaunchAt.ToString(culture)).Append(",\"impact\":").Append(step.ImpactAt.ToString(culture)).Append(",\"multiply\":").Append(step.Multiply ? "true" : "false").Append("}");
    }
    System.IO.File.WriteAllText(output + "Level-" + previewLevel + ".json", "{\"passed\":true,\"checks\":" + checks + ",\"level\":" + previewLevel + ",\"points\":\"" + model.Points.ToString(culture) + "\",\"duration\":" + sequence.Duration.ToString(culture) + ",\"operations\":[" + details + "]}\n");
